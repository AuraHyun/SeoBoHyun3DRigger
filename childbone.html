Child Bone Tool

3ds Max Max Script

Overview

Scale용 Child Bone을 쉽게 만들기 위한 스크립트입니다.

Tool Info

Software
3ds Max
Language
MaxScript

Script

ChildBone.ms 
-- Create Child Bone
(
	-- 시작점과 끝점이 꼭 필요
    local len = 10
	-- 실행 결과를 Ctrl + Z 한 번으로 전부 되돌릴 수 있게
    undo "Create Child Bones" on
    (
		-- 끝난 뒤 redrawViews()로 한 번만 그리도록
        with redraw off
        (
            for b in (selection as array) do
            (
                local c = boneSys.createBone [0,0,0] [len,0,0] [0,0,1]
                c.transform = b.transform
                c.parent = b
                c.name = "child " + b.name
            )
        )
    )
    redrawViews()
)

-- Set Scale Bulge
(
    -- 수정시
    local amount = 2.0
	-- 영향 범위로 키울수록 커짐
    local radius = 100.0

    -- 선택한 오브젝트들을 클릭한 순서대로 배열
    local sel = selection as array

    -- 움직일 컨트롤러
    local cursor = sel[1]

    -- 첫 번째 본부터 마지막 본까지 하나씩 반복
    for i = 2 to sel.count do
    (
        -- 지금 차례인 본
        local b = sel[i]

        -- 본 위치에 측정용 Point 생성, 이름은 "본이름 sample"
        local p = Point name:(b.name + " sample") pos:b.pos size:10

        -- Point를 본의 부모에 링크 -> 몸이 움직이면 같이 따라감
        -- 본 자체에 링크하면 스케일 때문에 순환 참조가 생겨서 부모에 붙임
        p.parent = b.parent

        -- 새 Scale Script 컨트롤러를 만들어 본의 스케일 컨트롤러로 교체
        local sc = Scale_Script()
        b.scale.controller = sc

        -- 스크립트 안에서 쓸 변수 등록
        sc.addNode "cursor" cursor          -- 컨트롤러 오브젝트
        sc.addNode "base" p                 -- 측정용 Point
        sc.addConstant "amount" amount      -- 최대 크기
        sc.addConstant "radius" radius      -- 영향 범위

        -- 매 프레임 계산될 식
        sc.script = "(
    local t = amin 1.0 ((distance base.pos cursor.pos) / radius)
	-- 선형 보간의 기본 공식
	-- a = 원래 크기, b = 최대
	-- 시작과 끝이 완만한 S자 모양
	-- 1.0 - 를 붙여서 방향을 뒤집음
	-- 3t² − 2t³은 그래픽스에서 smoothstep
	-- lerp(a, b, w) = a + (b - a) * w
    local w = 1.0 - t * t * (3.0 - 2.0 * t)
    local s = 1.0 + (amount - 1.0) * w
    [1.0, s, s]
	)"
    )
)

-- 값 조절을 원하는 경우
for o in selection do (o.scale.controller.setConstant "amount" 2.0
o.scale.controller.setConstant "radius" 80.0)
